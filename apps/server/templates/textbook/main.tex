\section{Kvadrat tenglama}

\begin{tarif}
  $ax^2 + bx + c = 0$ \ ($a \neq 0$) koʻrinishidagi tenglama
  \emph{kvadrat tenglama} deb ataladi.
\end{tarif}

\begin{teorema}[Ildizlar formulasi]
  Agar $D = b^2 - 4ac \geq 0$ boʻlsa, tenglamaning ildizlari
  \begin{equation}\label{eq:ildiz}
    x_{1,2} = \frac{-b \pm \sqrt{D}}{2a}.
  \end{equation}
\end{teorema}

\begin{misol}
  $x^2 - 5x + 6 = 0$ tenglamani yeching.
\end{misol}
\begin{yechim}
  Bu yerda $D = 25 - 24 = 1$, demak \eqref{eq:ildiz} formulaga koʻra $x_1 = 2$, $x_2 = 3$.
\end{yechim}

\section{Integral}

\muhim[Nyuton–Leybnits formulasi]{\int_a^b f(x)\,dx = F(b) - F(a)}

\[
  \int_0^1 x^2\,dx = \left.\frac{x^3}{3}\right|_0^1 = \frac{1}{3},
  \qquad
  \sum_{k=1}^{n} k = \frac{n(n+1)}{2}.
\]
