% Medal: \medal{rang}{I}
\newcommand\medal[2]{%
  \begin{tikzpicture}[baseline=(c.center)]
    % Lentalar / ribbons
    \fill[dQizil] (-0.55,-0.2) -- (-0.95,-2.0) -- (-0.6,-1.75) -- (-0.35,-2.05) -- (0,-0.3) -- cycle;
    \fill[dQizil!80!black] (0.55,-0.2) -- (0.95,-2.0) -- (0.6,-1.75) -- (0.35,-2.05) -- (0,-0.3) -- cycle;
    % Dafna shoxlari / laurel branches on both sides
    \draw[#1!70!black, line width=1pt] (-120:1.75) arc[start angle=240, end angle=120, radius=1.75];
    \draw[#1!70!black, line width=1pt] (-60:1.75) arc[start angle=-60, end angle=60, radius=1.75];
    % Barglar shox boʻylab yuqoriga qaragan / leaves lean up along the branch
    \foreach \a in {-45,-25,...,55} {
      \fill[#1!80!black, shift={(\a:1.75)}, rotate=\a+45] (0.22,0) ellipse (0.25 and 0.09);
      \fill[#1!80!black, shift={(\a:1.75)}, rotate=\a+135] (0.2,0) ellipse (0.21 and 0.08);}
    \foreach \a in {125,145,...,225} {
      \fill[#1!80!black, shift={(\a:1.75)}, rotate=\a-45] (0.22,0) ellipse (0.25 and 0.09);
      \fill[#1!80!black, shift={(\a:1.75)}, rotate=\a-135] (0.2,0) ellipse (0.21 and 0.08);}
    \node[circle, fill=#1, draw=#1!60!black, line width=1.5pt, minimum size=2.4cm,
      font=\sarlavhashrift\bfseries\fontsize{30}{30}\selectfont, text=white] (c) {#2};
  \end{tikzpicture}}
